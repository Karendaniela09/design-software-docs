-- Evidencia de consulta de solo lectura realizada por el aprendiz en PostgreSQL 16 local.
-- Contenedor: env-integ-jsgh1-postgres-1. Datos de laboratorio, NO ambiente compartido.
SELECT
    e.name AS ambiente,
    m.start_date, m.start_time, m.end_time,
    m.reason, m.reason_detail,
    CASE WHEN m.reason_detail = m.reason THEN 'OK' ELSE 'REVISAR' END AS r6_sync,
    CASE WHEN EXISTS (
        SELECT 1 FROM training_environment.outbox o
        WHERE o.event_type = 'environment.maintenance.started'
          AND o.payload::jsonb -> 'payload' ->> 'maintenance_id' = m.id::text
    ) THEN 'OK' ELSE 'FALTA' END AS outbox_event
FROM training_environment.maintenance m
JOIN training_environment.environment e ON e.id = m.environment_id
WHERE e.name = 'Aula 101' AND m.reason LIKE 'PRUEBA_UI_R6_20261020%'
ORDER BY m.created_at DESC LIMIT 1;
